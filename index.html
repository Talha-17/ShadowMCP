import re
import json
import time
from fastapi import FastAPI, Request, HTTPException
from fastapi.responses import JSONResponse
import httpx

app = FastAPI(
    title="ShadowMCP Gateway",
    description="Enterprise Security Middleware for Model Context Protocol (MCP)",
    version="2.4.0"
)

UPSTREAM_FASTMCP_URL = "http://localhost:8001/mcp"

# Security Policy Pattern Definitions
SECURITY_POLICIES = {
    "PATH_TRAVERSAL": [r"\.\./", r"/etc/passwd", r"\.ssh", r"\.env", r"C:\\Windows"],
    "PROMPT_INJECTION": [r"ignore previous instructions", r"system override", r"forget guidelines", r"print hidden prompt"],
    "UNRESTRICTED_SHELL": [r";\s*rm\s+-rf", r"\|\s*bash", r"nc\s+-e", r"curl\s+.*\|\s*sh"],
    "SECRET_EXFILTRATION": [r"AKIA[0-9A-Z]{16}", r"sk-[a-zA-Z0-9]{32,}", r"-----BEGIN PRIVATE KEY-----"]
}

def inspect_jsonrpc_payload(payload: dict) -> tuple[bool, str, int]:
    """Inspects JSON-RPC tool parameters against security rules with zero-copy regex matching."""
    params_str = json.dumps(payload.get("params", {}))
    
    for category, patterns in SECURITY_POLICIES.items():
        for pattern in patterns:
            if re.search(pattern, params_str, re.IGNORECASE):
                return True, f"{category} Violation (Pattern: '{pattern}')", 96
                
    return False, "Payload Validated", 0

@app.post("/mcp/rpc")
async def intercept_and_proxy_rpc(request: Request):
    start_time = time.perf_counter()
    
    try:
        payload = await request.json()
    except Exception:
        raise HTTPException(status_code=400, detail="Invalid JSON-RPC Format")

    is_threat, reason, risk_score = inspect_jsonrpc_payload(payload)

    if is_threat:
        latency_ms = (time.perf_counter() - start_time) * 1000
        return JSONResponse(
            status_code=403,
            content={
                "jsonrpc": "2.0",
                "id": payload.get("id"),
                "error": {
                    "code": -32600,
                    "message": f"ShadowMCP Gateway Interdiction: {reason}",
                    "risk_score": risk_score,
                    "inspection_time_ms": round(latency_ms, 3)
                }
            }
        )

    # Forward validated payload to upstream tool server
    async with httpx.AsyncClient() as client:
        try:
            upstream_res = await client.post(UPSTREAM_FASTMCP_URL, json=payload, timeout=10.0)
            return JSONResponse(content=upstream_res.json(), status_code=upstream_res.status_code)
        except httpx.RequestError as exc:
            raise HTTPException(status_code=502, detail=f"Upstream FastMCP Server Unreachable: {exc}")

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=8000)
